- Create in AppServiceProver, Apply it in the routes directly
use Illuminate\Cache\RateLimiting\Limit;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\RateLimiter;

/**
 * Bootstrap any application services.
 */
public function boot(): void
{
    RateLimiter::for('login', function (Request $request) {
        return Limit::perMinute(60)->by($request->user()?->id ?: $request->ip());
    });
}

প্রোডাকশন সিকিউরিটির জন্য আরও ৩টি জরুরি পদক্ষেপ
আপনার অ্যাপ্লিকেশনের রেসপন্স হেডারে কিছু সিকিউরিটি প্যারামিটার যোগ করা উচিত। এটি AppServiceProvider বা একটি আলাদা Middleware এ করতে পারেন:
X-Frame-Options: ক্লিকজ্যাকিং (Clickjacking) প্রতিরোধে।
X-Content-Type-Options: ব্রাউজারকে ফাইল টাইপ গেস করা থেকে বিরত রাখতে।
Content-Security-Policy (CSP): ক্রস-সাইট স্ক্রিপ্টিং (XSS) প্রতিরোধে।
